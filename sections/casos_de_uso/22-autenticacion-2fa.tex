\subsubsection{Verificar segundo factor luego del login inicial}

\textbf{Descripción:} Tras credenciales válidas, si el usuario tiene 2FA activo se solicita código TOTP (aplicación autenticadora). Se verifica y se completa la sesión.

\vspace{0.5em}
\noindent\textbf{PreCondiciones:}
\begin{itemize}[nosep]
    \item Usuario activo y aprobado.
    \item 2FA habilitado (secret almacenado) para ese usuario.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{PostCondiciones:}
\begin{itemize}[nosep]
    \item Sesión autenticada de segundo factor; acceso normal concedido.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Actores:} Observador, Analista, Administrador.

\vspace{0.5em}
\noindent\textbf{Flujo de eventos principal:}
\begin{enumerate}[nosep]
    \item Usuario envía credenciales al endpoint de login.
    \item Backend valida usuario y estado (no pendiente) y detecta 2FA.
    \item Responde solicitando código TOTP.
    \item Usuario envía código (\texttt{Verify2FARequestDTO}).
    \item Backend valida contra \texttt{TotpService}.
    \item Si es correcto, retorna token final y habilita acceso.
\end{enumerate}

\vspace{0.5em}
\noindent\textbf{Flujos alternativos:}

\vspace{0.3em}
\noindent\textit{A1. Código incorrecto}
\begin{itemize}[nosep]
    \item Respuesta de error, se permite reintento.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A2. Usuario sin 2FA}
\begin{itemize}[nosep]
    \item Emisión de token sin paso adicional.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A3. Usuario pendiente de aprobación}
\begin{itemize}[nosep]
    \item Se bloquea (ver siguiente caso).
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Requerimientos especiales:}
\begin{itemize}[nosep]
    \item Validación TOTP en servicio dedicado.
    \item Control por controlador específico.
    \item Email informativo al activar 2FA.
\end{itemize}
\end{itemize}
