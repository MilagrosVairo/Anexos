\subsubsection{Aprobar, Rechazar y Gestionar Usuarios}

\textbf{Descripción:} El Administrador revisa solicitudes de usuarios pendientes, aprueba o rechaza el acceso, y gestiona usuarios existentes cambiando roles o estados.

\vspace{0.5em}
\noindent\textbf{PreCondiciones:}
\begin{itemize}[nosep]
    \item El usuario debe tener rol \texttt{ADMIN}.
    \item Para aprobar/rechazar: debe existir al menos una solicitud pendiente.
    \item Para gestionar: debe existir el usuario en el sistema.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{PostCondiciones:}
\begin{itemize}[nosep]
    \item Aprobar: Usuario pasa a estado \texttt{ACTIVO} con rol asignado.
    \item Rechazar: Usuario eliminado del sistema.
    \item Gestionar: Cambios aplicados (rol, estado activo/inactivo).
    \item Notificaciones y emails enviados automáticamente.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Actores:} Administrador.

\vspace{0.5em}
\noindent\textbf{Flujo de eventos principal:}
\begin{enumerate}[nosep]
    \item \textbf{Aprobar Usuario}
    \begin{enumerate}[nosep]
        \item \textit{Revisión de solicitudes}
        \begin{itemize}[nosep]
            \item El Administrador accede a ``Gestión de Usuarios''.
            \item Selecciona pestaña ``Solicitudes Pendientes''.
            \item Ver lista de usuarios con estado \texttt{PENDIENTE}:
            \begin{itemize}[nosep]
                \item Nombre completo
                \item Nombre de usuario
                \item Email
                \item Fecha de registro
            \end{itemize}
            \item Puede buscar/filtrar solicitudes.
        \end{itemize}

        \item \textit{Aprobación}
        \begin{itemize}[nosep]
            \item Selecciona una solicitud para revisar.
            \item Ve el detalle completo del solicitante.
            \item Selecciona ``Aprobar usuario''.
            \item El sistema muestra formulario de aprobación:
            \begin{itemize}[nosep]
                \item Seleccionar rol: \texttt{ANALISTA} o \texttt{ADMIN}
            \end{itemize}
            \item El Administrador selecciona el rol apropiado.
            \item Confirma la aprobación.
            \item El sistema:
            \begin{itemize}[nosep]
                \item Cambia estado a \texttt{ACTIVO}.
                \item Asigna el rol seleccionado.
                \item Establece \texttt{activo = true}.
                \item Crea notificación para el usuario.
                \item Envía email de bienvenida al usuario.
                \item Muestra confirmación: ``Usuario aprobado exitosamente''.
            \end{itemize}
        \end{itemize}
    \end{enumerate}

    \item \textbf{Rechazar Usuario}
    \begin{itemize}[nosep]
        \item El Administrador accede a una solicitud pendiente.
        \item Revisa los datos.
        \item Selecciona ``Rechazar solicitud''.
        \item El sistema muestra confirmación:
        \begin{itemize}[nosep]
            \item ``¿Está seguro de rechazar esta solicitud?''
            \item ``El usuario será eliminado del sistema''
        \end{itemize}
        \item El Administrador confirma.
        \item El sistema:
        \begin{itemize}[nosep]
            \item Crea notificación de rechazo.
            \item Elimina el usuario del sistema.
            \item Muestra confirmación: ``Solicitud rechazada''.
        \end{itemize}
    \end{itemize}

    \item \textbf{Gestionar Usuario Existente}
    \begin{itemize}[nosep]
        \item El Administrador accede a ``Gestión de Usuarios''.
        \item Selecciona pestaña ``Todos los Usuarios''.
        \item Puede filtrar por:
        \begin{itemize}[nosep]
            \item Rol (\texttt{ANALISTA}, \texttt{ADMIN})
            \item Estado (Activo, Inactivo)
            \item Búsqueda por nombre, usuario o email
        \end{itemize}
        \item Selecciona un usuario para gestionar.
        \item Puede modificar:
        \begin{itemize}[nosep]
            \item Rol: cambiar entre \texttt{ANALISTA} y \texttt{ADMIN}
            \item Estado activo: activar/desactivar usuario
        \end{itemize}
        \item Guarda los cambios.
        \item El sistema:
        \begin{itemize}[nosep]
            \item Actualiza el usuario.
            \item Sincroniza estado con campo \texttt{activo}.
            \item Registra el cambio.
            \item Muestra confirmación.
        \end{itemize}
    \end{itemize}
\end{enumerate}

\vspace{0.5em}
\noindent\textbf{Flujos alternativos:}

\vspace{0.3em}
\noindent\textit{A1. Intentar aprobar con email duplicado}
\begin{itemize}[nosep]
    \item El sistema detecta que el email ya existe en otro usuario activo.
    \item Muestra error: ``El email ya está en uso por otro usuario activo''.
    \item No permite aprobar.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A2. Cancelar aprobación/rechazo}
\begin{itemize}[nosep]
    \item El usuario cierra el diálogo.
    \item No se realiza ningún cambio.
    \item La solicitud permanece \texttt{PENDIENTE}.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A3. Desactivar usuario con sesión activa}
\begin{itemize}[nosep]
    \item El sistema permite desactivar.
    \item El usuario pierde acceso en su próximo intento de login.
    \item Las sesiones activas pueden continuar hasta expirar.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A4. Cambiar rol de ADMIN a ANALISTA}
\begin{itemize}[nosep]
    \item El sistema verifica que haya al menos otro \texttt{ADMIN} activo.
    \item Si es el último \texttt{ADMIN} $\rightarrow$ muestra advertencia.
    \item Requiere confirmación adicional.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A5. Error al enviar email}
\begin{itemize}[nosep]
    \item La aprobación/rechazo se completa exitosamente.
    \item El email se intenta enviar pero no bloquea el proceso.
    \item Se registra log del error.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Requerimientos especiales:}
\begin{itemize}[nosep]
    \item Solo Administradores pueden gestionar usuarios.
    \item Email de bienvenida automático al aprobar.
    \item Notificación automática al aprobar/rechazar.
    \item Validación de al menos un \texttt{ADMIN} activo en el sistema.
    \item Búsqueda y filtrado eficiente con paginación.
    \item Sincronización automática entre estado y campo \texttt{activo}.
\end{itemize}
