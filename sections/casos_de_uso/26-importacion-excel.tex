\subsubsection{Importar datos históricos desde un archivo Excel a la entidad Legado}

\textbf{Descripción:} El sistema procesa un archivo Excel, mapea filas a entidades Legado (incluyendo \texttt{filaExcel}) y persiste los registros, permitiendo carga de datos previos externos.

\vspace{0.5em}
\noindent\textbf{PreCondiciones:}
\begin{itemize}[nosep]
   \item Archivo Excel válido suministrado.
   \item Formato esperado (columnas reconocibles) sin corrupción.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{PostCondiciones:}
\begin{itemize}[nosep]
   \item Registros Legado creados o ignorados según modo (simulación vs persistencia real) y resultado con conteos.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Actores:} Administrador.

\vspace{0.5em}
\noindent\textbf{Flujo de eventos principal:}
\begin{enumerate}[nosep]
   \item Usuario sube archivo y elige simulación (dry-run) o importación definitiva.
   \item Servicio lee workbook y recorre filas válidas.
   \item Para cada fila crea Legado, asigna campos y \texttt{filaExcel}.
   \item Si modo definitivo: guarda vía repositorio; si simulación: acumula sin persistir.
   \item Construye resultado con filas procesadas, guardadas y errores.
   \item Retorna resultado al usuario.
\end{enumerate}

\vspace{0.5em}
\noindent\textbf{Flujos alternativos:}

\vspace{0.3em}
\noindent\textit{A1. Archivo vacío}
\begin{itemize}[nosep]
   \item Resultado con contadores en cero.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A2. Error de formato (columna obligatoria faltante)}
\begin{itemize}[nosep]
   \item Se registra error y puede abortar o continuar según implementación.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A3. Dry-run}
\begin{itemize}[nosep]
   \item No se guarda nada; solo se reporta lo que se importaría.
\end{itemize}

\vspace{0.3em}
\noindent\textit{A4. Fila con datos inválidos}
\begin{itemize}[nosep]
   \item Se omite e incrementa contador de errores.
\end{itemize}

\vspace{0.5em}
\noindent\textbf{Requerimientos especiales:}
\begin{itemize}[nosep]
   \item Campo filaExcel para rastreo de origen.
\end{itemize}
